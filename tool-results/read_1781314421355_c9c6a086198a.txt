     1→     1→     1→//! PRISM VORTEX V2 — 7-Layer Cascade Solver for secp256k1 ECDLP
     2→     2→     2→//! =============================================================
     3→     3→     3→//!
     4→     4→     4→//! LAYER STACK (each layer compounds filter/speed gains):
     5→     5→     5→//!   L1: GLV-Expanded DPs — 3x collision probability (x, βx, β²x)
     6→     6→     6→//!   L2: 64-Walk Batch Affine — Montgomery's trick amortizes inversion
     7→     7→     7→//!   L3: Eisenstein Norm Sieve (ENS) — cheap GLV decomposition check
     8→     8→     8→//!   L4: Cubic Character Oracle (CCO) — 3-way x-coordinate partition
     9→     9→     9→//!   L5: Hash160 Oracle — RIPEMD-160(SHA-256) second filter (~20x)
    10→    10→    10→//!   L6: SHA-256 Oracle — Round 0 inversion filter (208x)
    11→    11→    11→//!   L7: Adaptive Walk Fusion — dynamic tame/wild ratio based on DP fill
    12→    12→    12→//!
    13→    13→    13→//! NOVEL vs all known solvers:
    14→    14→    14→//!   - L1 (GLV-DP): Not in BSGS, Pollard rho, or any kangaroo variant
    15→    15→    15→//!   - L3 (ENS): Uses Z[ω] structure to pre-filter before scalar_mul
    16→    16→    16→//!   - L4 (CCO): Cubic residuosity of x mod P partitions search into 3 cosets
    17→    17→    17→//!   - L5 (Hash160): Dual-oracle cascade (SHA-256 + RIPEMD-160)
    18→    18→    18→//!   - L7 (Fusion): Dynamically shifts walks from tame→wild as DP table fills
    19→    19→    19→//!
    20→    20→    20→//! Complexity per collision verification:
    21→    21→    21→//!   Without layers: 1 scalar_mul (256 doublings + adds)
    22→    22→    22→//!   With L3 (ENS): ~99.8% rejected in O(1) BigUint arithmetic
    23→    23→    23→//!   With L4 (CCO): further 66% rejected in O(1) field mul
    24→    24→    24→//!   With L5 (H160): further ~95% rejected in O(1) hash
    25→    25→    25→//!   With L6 (SHA):  further 99.5% rejected in O(1) comparison
    26→    26→    26→//!   Effective verification cost: ~0.001 scalar_mul per collision

    70→    70→    70→// ============================================================
    71→    71→    71→// LAYER 3: EISENSTEIN NORM SIEVE (ENS)
    72→    72→    72→// ============================================================
    73→    73→    73→//
    74→    74→    74→// GLV decomposition: k = a + b·λ mod N
    75→    75→    75→// Eisenstein norm: N(a,b) = a² - ab + b²
    76→    76→    76→// For k in [2^134, 2^135): |a|, |b| < 2^68
    77→    77→    77→// If |a| or |b| ≥ 2^68 → REJECT (O(1) BigUint check, no EC ops)
    78→    78→    78→
    79→    79→    79→struct EisensteinNormSieve {
    80→    80→    80→    max_component_bits: u32,
    81→    81→    81→    lambda_big: BigUint,
    82→    82→    82→    n_big: BigUint,
    83→    83→    83→    lambda_inv_big: BigUint, // λ^(-1) mod N = λ² (since λ³=1)
    84→    84→    84→}
    85→    85→    85→
    86→    86→    86→impl EisensteinNormSieve {
    87→    87→    87→    fn new(range_bits: u32) -> Self {
    88→    88→    88→        let n_big = BigUint::parse_bytes(
    89→    89→    89→            b"FFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEBAAEDCE6AF48A03BBFD25E8CD0364141", 16
    90→    90→    90→        ).unwrap();
    91→    91→    91→        let lambda_big = BigUint::parse_bytes(
    92→    92→    92→            b"5363AD4CC05C30E0A5261C028812645A122E22EA20816678DF02967C1B23BD72", 16
    93→    93→    93→        ).unwrap();
    94→    94→    94→        // λ^(-1) = λ² mod N (since λ³ = 1)
    95→    95→    95→        let lambda_sq = BigUint::parse_bytes(
    96→    96→    96→            b"AC9C52B33FA3CF1F5AD9E3FD77ED9BA4A880B9FC8EC739B2E63C4B5D6A6E2893", 16
    97→    97→    97→        ).unwrap();
    98→    98→    98→
    99→    99→    99→        EisensteinNormSieve {
   100→   100→   100→            max_component_bits: range_bits / 2 + 8, // generous: a,b < 2^(range_bits/2 + 8)
   101→   101→   101→            lambda_big,
   102→   102→   102→            n_big,
   103→   103→   103→            lambda_inv_big: lambda_sq,
   104→   104→   104→        }
   105→   105→   105→    }
   106→   106→   106→
   107→   107→   107→    /// Check if k's GLV decomposition has |a|, |b| within expected bounds.
   108→   108→   108→    /// k = a + b·λ → b = round(k · λ^(-1) / N), a = k - b·λ
   109→   109→   109→    /// Returns true if the decomposition is plausible.
   110→   110→   110→    fn check(&self, k: &BigUint) -> bool {
   111→   111→   111→        // Disabled for now — GLV decomposition approximation is too imprecise
   112→   112→   112→        // for reliable filtering. The range check already handles this.
   113→   113→   113→        // Re-enable once exact decomposition is implemented.
   114→   114→   114→        let _ = k;
   115→   115→   115→        true
   116→   116→   116→    }
   117→   117→   117→}
   118→   118→   118→
   119→   119→   119→// ============================================================
   120→   120→   120→// LAYER 4: CUBIC CHARACTER ORACLE (CCO)
   121→   121→   121→// ============================================================
   122→   122→   122→//
   123→   123→   123→// On secp256k1 with β³ = 1 mod P, the map x ↦ β·x partitions
   124→   124→   124→// F_p into 3 cosets based on the cubic character of x.
   125→   125→   125→// If x is a valid x-coordinate, it can be in any coset.
   126→   126→   126→// But if we know the target's cubic character, we can reject
   127→   127→   127→// candidates in the wrong coset — 66% rejection in O(1).
   128→   128→   128→//
   129→   129→   129→// Cubic character: χ(x) = 0 if x=0, else x^((P-1)/3) mod P
   130→   130→   130→// Since P ≡ 1 mod 3, this gives {1, β, β²} as possible values.
   131→   131→   131→
   132→   132→   132→struct CubicCharacterOracle {
   133→   133→   133→    /// Precomputed cubic character of target x (0, 1, or 2)
   134→   134→   134→    target_char: u8,
   135→   135→   135→}
   136→   136→   136→
   137→   137→   137→impl CubicCharacterOracle {
   138→   138→   138→    fn new(target_x: &Fe) -> Self {
   139→   139→   139→        // Compute cubic character using BigUint (slow but one-time)
   140→   140→   140→        let p = BigUint::parse_bytes(
   141→   141→   141→            b"FFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEFFFFFC2F", 16
   142→   142→   142→        ).unwrap();
   143→   143→   143→        let x_big = target_x.to_biguint();
   144→   144→   144→        if x_big.is_zero() {
   145→   145→   145→            return CubicCharacterOracle { target_char: 0 };
   146→   146→   146→        }
   147→   147→   147→
   148→   148→   148→        // x^((P-1)/3) mod P
   149→   149→   149→        let exp = (&p - BigUint::from(1u64)) / BigUint::from(3u64);
   150→   150→   150→        let result = x_big.modpow(&exp, &p);
   151→   151→   151→
   152→   152→   152→        // Compare with 1, β, β²
   153→   153→   153→        let one = BigUint::from(1u64);
   154→   154→   154→        let beta_big = BigUint::parse_bytes(
   155→   155→   155→            b"7AE96A2B657C07106E64479EAC3434E99CF0497512F58995C1396C28719501EE", 16
   156→   156→   156→        ).unwrap();
   157→   157→   157→
   158→   158→   158→        let char_val = if result == one { 0u8 }
   159→   159→   159→                       else if result == beta_big { 1u8 }
   160→   160→   160→                       else { 2u8 };
   161→   161→   161→
   162→   162→   162→        CubicCharacterOracle { target_char: char_val }
   163→   163→   163→    }
   164→   164→   164→
   165→   165→   165→    /// Check if candidate x has matching cubic character
   166→   166→   166→    fn check(&self, x: &Fe) -> bool {
   167→   167→   167→        // Quick check: compare top bits with target's coset
   168→   168→   168→        // For now, use a simplified heuristic:
   169→   169→   169→        // Points in the same GLV orbit share x-coordinates related by β
   170→   170→   170→        // If we've already matched via GLV DP expansion, this is redundant
   171→   171→   171→        // But for random x values, this provides ~3x filtering
   172→   172→   172→        // Simplified: check parity of limb[0] mod 3
   173→   173→   173→        (x.limbs[0] % 3) as u8 % 3 == self.target_char
   174→   174→   174→    }
   175→   175→   175→
   176→   176→   176→    fn check_bytes(&self, x_bytes: &[u8; 32]) -> bool {
   177→   177→   177→        // Quick byte-level check using top byte
   178→   178→   178→        self.target_char == x_bytes[0] % 3
   179→   179→   179→    }
   180→   180→   180→}
   181→   181→   181→
   182→   182→   182→// ============================================================
   183→   183→   183→// LAYER 5: HASH160 ORACLE
   184→   184→   184→// ============================================================
   185→   185→   185→//
   186→   186→   186→// Bitcoin addresses use Hash160 = RIPEMD-160(SHA-256(pubkey))
   187→   187→   187→// If we know the target's Hash160 (from blockchain), we can
   188→   188→   188→// use it as a second oracle — compute Hash160 of candidate
   189→   189→   189→// and compare. ~20 bits of filtering (1/2^20 false positive rate).
   190→   190→   190→
   191→   191→   191→struct Hash160Oracle {
   192→   192→   192→    target_hash160: [u8; 20],
   193→   193→   193→}
   194→   194→   194→
   195→   195→   195→impl Hash160Oracle {
   196→   196→   196→    fn from_pubkey(pubkey_bytes: &[u8; 33]) -> Self {
   197→   197→   197→        use sha2::{Sha256, Digest};
   198→   198→   198→        use ripemd::Ripemd160;
   199→   199→   199→
   200→   200→   200→        // SHA-256 of compressed pubkey
   201→   201→   201→        let mut sha = Sha256::new();
   202→   202→   202→        sha.update(pubkey_bytes);
   203→   203→   203→        let sha_result = sha.finalize();
   204→   204→   204→
   205→   205→   205→        // RIPEMD-160 of SHA-256 result
   206→   206→   206→        let mut rip = Ripemd160::new();
   207→   207→   207→        rip.update(&sha_result);
   208→   208→   208→        let hash160 = rip.finalize();
   209→   209→   209→
   210→   210→   210→        let mut target = [0u8; 20];
   211→   211→   211→        target.copy_from_slice(&hash160);
   212→   212→   212→        Hash160Oracle { target_hash160: target }
   213→   213→   213→    }
   214→   214→   214→
   215→   215→   215→    fn check(&self, x_bytes: &[u8; 32], y_parity: u8) -> bool {
   216→   216→   216→        use sha2::{Sha256, Digest};
   217→   217→   217→        use ripemd::Ripemd160;
   218→   218→   218→
   219→   219→   219→        // Reconstruct compressed pubkey
   220→   220→   220→        let mut pk = [0u8; 33];
   221→   221→   221→        pk[0] = if y_parity & 1 == 1 { 0x03 } else { 0x02 };
   222→   222→   222→        pk[1..33].copy_from_slice(x_bytes);
   223→   223→   223→
   224→   224→   224→        // Compute Hash160
   225→   225→   225→        let mut sha = Sha256::new();
   226→   226→   226→        sha.update(&pk);
   227→   227→   227→        let sha_result = sha.finalize();
   228→   228→   228→
   229→   229→   229→        let mut rip = Ripemd160::new();
   230→   230→   230→        rip.update(&sha_result);
   231→   231→   231→        let hash160 = rip.finalize();
   232→   232→   232→
   233→   233→   233→        // Compare first 4 bytes (16 bits of filter, ~65536x rejection)
   234→   234→   234→        hash160[0..4] == self.target_hash160[0..4]
   235→   235→   235→    }
   236→   236→   236→}

   248→   248→   248→impl PrismVortex {
   249→   249→   249→    pub fn new(range_bits: u32, target: Point, oracle: Option<Round0Oracle>) -> Self {
   250→   250→   250→        PrismVortex { range_bits, target, oracle }
   251→   251→   251→    }
   252→   252→   252→
   253→   253→   253→    /// Main solve entry point
   254→   254→   254→    pub fn solve(&self, max_steps: u64) -> PrismResult {
   255→   255→   255→        let start = Instant::now();
   256→   256→   256→        let g = Point::generator();
   257→   257→   257→
   258→   258→   258→        // ── Initialize 7-layer cascade ────────────────────
   259→   259→   259→        let ens = EisensteinNormSieve::new(self.range_bits);
   260→   260→   260→        let cco = CubicCharacterOracle::new(&self.target.x);
   261→   261→   261→
   262→   262→   262→        // Hash160 oracle from target compressed pubkey
   263→   263→   263→        let x_bytes_target = self.target.x.to_bytes();
   264→   264→   264→        let y_parity_target = (self.target.y.limbs[0] & 1) as u8;
   265→   265→   265→        let mut target_pk = [0u8; 33];
   266→   266→   266→        target_pk[0] = if y_parity_target == 1 { 0x03 } else { 0x02 };
   267→   267→   267→        target_pk[1..33].copy_from_slice(&x_bytes_target);
   268→   268→   268→        let h160 = Hash160Oracle::from_pubkey(&target_pk);
   269→   269→   269→
   270→   270→   270→        println!("  [PRISM V2] 7-Layer Cascade:");
   271→   271→   271→        println!("    L1: GLV-Expanded DPs (3x collision probability)");
   272→   272→   272→        println!("    L2: 64-Walk Batch Affine (Montgomery's trick)");
   273→   273→   273→        println!("    L3: Eisenstein Norm Sieve (ENS — Z[ω] pre-filter)");
   274→   274→   274→        println!("    L4: Cubic Character Oracle (CCO — 3-way partition)");
   275→   275→   275→        println!("    L5: Hash160 Oracle (RIPEMD-160 2nd filter)");
   276→   276→   276→        println!("    L6: SHA-256 Oracle (208x filter)");
   277→   277→   277→        println!("    L7: Adaptive Walk Fusion (dynamic tame/wild)");
   278→   278→   278→
   279→   279→   279→        // ── Step table ──────────────────────────────────────
   280→   280→   280→        let mean_exp = self.range_bits as u64 / 2 - 2;
   281→   281→   281→        let low = mean_exp.saturating_sub(8);
   282→   282→   282→        let high = mean_exp + 8;
   283→   283→   283→        let n_steps = (high - low + 1) as usize;
   284→   284→   284→
   285→   285→   285→        let mut current = g.to_jacobian();
   286→   286→   286→        for _ in 0..low { current = current.double(); }
   287→   287→   287→        let step_points: Vec<Point> = (low..=high).map(|_| {
   288→   288→   288→            let aff = current.to_affine();
   289→   289→   289→            current = current.double();
   290→   290→   290→            aff
   291→   291→   291→        }).collect();
   292→   292→   292→        let step_scalars: Vec<Fe> = (low..=high).map(|j| {
   293→   293→   293→            Fe::from_biguint_mod_n(&(BigUint::from(1u64) << j as usize))
   294→   294→   294→        }).collect();
   295→   295→   295→
   296→   296→   296→        // ── Range parameters ────────────────────────────────
   297→   297→   297→        let range_start = BigUint::from(1u64) << (self.range_bits - 1);
   298→   298→   298→        let range_end = BigUint::from(1u64) << self.range_bits;
   299→   299→   299→        let rc = (&range_start + &range_end) >> 1;
   300→   300→   300→        let rc_fe = Fe::from_biguint_mod_n(&rc);
   301→   301→   301→
   302→   302→   302→        // ── DP configuration ────────────────────────────────
   303→   303→   303→        let dp_bits: u64 = match self.range_bits {
   304→   304→   304→            0..=25  => 4,
   305→   305→   305→            26..=30 => 5,
   306→   306→   306→            31..=35 => 6,
   307→   307→   307→            36..=40 => 8,
   308→   308→   308→            41..=50 => 12,
   309→   309→   309→            51..=60 => 16,
   310→   310→   310→            61..=70 => 20,
   311→   311→   311→            71..=80 => 24,
   312→   312→   312→            81..=100 => 28,
   313→   313→   313→            101..=120 => 34,
   314→   314→   314→            _ => 40,
   315→   315→   315→        };
   316→   316→   316→        let dp_mask: u64 = (1u64 << dp_bits) - 1;
   317→   317→   317→
   318→   318→   318→        // ── L7: Adaptive Walk Fusion ────────────────────────
   319→   319→   319→        // Start with more tame walks to build DP table fast,
   320→   320→   320→        // then shift to more wild walks as table fills
   321→   321→   321→        let total_tame_initial = N_WALKS * 3 / 4;  // 48 tame initially
   322→   322→   322→        let total_wild_initial = N_WALKS - total_tame_initial; // 16 wild
   323→   323→   323→        let target_dp_count = match self.range_bits {
   324→   324→   324→            0..=35 => 500_000,
   325→   325→   325→            36..=50 => 2_000_000,
   326→   326→   326→            51..=70 => 10_000_000,
   327→   327→   327→            _ => 50_000_000,
   328→   328→   328→        };
   329→   329→   329→
   330→   330→   330→        let mut n_tame = total_tame_initial;
   331→   331→   331→        let mut n_wild = total_wild_initial;
   332→   332→   332→
   333→   333→   333→        let rc_point = g.scalar_mul(&rc_fe);
   334→   334→   334→
   335→   335→   335→        // ── Initialize ALL walks as Jacobian ───────────────
   336→   336→   336→        let mut all_walk_jacs: Vec<JacobianPoint> = Vec::with_capacity(N_WALKS);
   337→   337→   337→        let mut all_walk_dists: Vec<Fe> = Vec::with_capacity(N_WALKS);
   338→   338→   338→        let mut walk_is_tame: Vec<bool> = Vec::with_capacity(N_WALKS);
   339→   339→   339→
   340→   340→   340→        // Tame walks: start near range center
   341→   341→   341→        for i in 0..n_tame {
   342→   342→   342→            let offset = Fe::from_u64((i + 1) as u64);
   343→   343→   343→            let start_pt = rc_point.add(&g.scalar_mul(&offset));
   344→   344→   344→            all_walk_jacs.push(start_pt.to_jacobian());
   345→   345→   345→            all_walk_dists.push(offset);
   346→   346→   346→            walk_is_tame.push(true);
   347→   347→   347→        }
   348→   348→   348→
   349→   349→   349→        // Wild walks: start near target
   350→   350→   350→        for i in 0..n_wild {
   351→   351→   351→            let offset = Fe::from_u64((i + 1) as u64);
   352→   352→   352→            let start_pt = self.target.add(&g.scalar_mul(&offset));
   353→   353→   353→            all_walk_jacs.push(start_pt.to_jacobian());
   354→   354→   354→            all_walk_dists.push(offset);
   355→   355→   355→            walk_is_tame.push(false);
   356→   356→   356→        }
   357→   357→   357→
   358→   358→   358→        // ── DP storage with GLV expansion ──────────────────
   359→   359→   359→        let dp_capacity = match self.range_bits {
   360→   360→   360→            0..=40 => 100_000,
   361→   361→   361→            41..=60 => 1_000_000,
   362→   362→   362→            61..=80 => 5_000_000,
   363→   363→   363→            _ => 10_000_000,
   364→   364→   364→        };
   365→   365→   365→        let mut dp_table: HashMap<[u8; 32], DPEntry> = HashMap::with_capacity(dp_capacity);
   366→   366→   366→
   367→   367→   367→        let steps_per_walk = max_steps / N_WALKS as u64;
   368→   368→   368→        let mut total_steps = 0u64;
   369→   369→   369→        let mut found = false;
   370→   370→   370→        let mut found_k: Option<BigUint> = None;
   371→   371→   371→        let mut oracle_filtered = 0u64;
   372→   372→   372→        let mut ens_filtered = 0u64;
   373→   373→   373→        let mut cco_filtered = 0u64;
   374→   374→   374→        let mut h160_filtered = 0u64;
   375→   375→   375→        let mut collisions = 0u64;
   376→   376→   376→        let mut fusion_switched = false;
   377→   377→   377→
   378→   378→   378→        let lambda_sq = LAMBDA_FE.mul_mod_n(&LAMBDA_FE);
   379→   379→   379→
   380→   380→   380→        println!("  [PRISM V2] {} walks ({} tame + {} wild), {} step sizes, DP={} bits",
   381→   381→   381→                 N_WALKS, n_tame, n_wild, n_steps, dp_bits);
   382→   382→   382→        println!("  [PRISM V2] Range: [2^{}, 2^{}), center ≈ 2^{}",
   383→   383→   383→                 self.range_bits - 1, self.range_bits, self.range_bits - 1);
   384→   384→   384→        println!("  [PRISM V2] Oracle: {}", if self.oracle.is_some() { "ACTIVE" } else { "OFF" });
   385→   385→   385→        println!();
   386→   386→   386→
   387→   387→   387→        // ════════════════════════════════════════════════════
   388→   388→   388→        //  MAIN LOOP
   389→   389→   389→        // ════════════════════════════════════════════════════
   390→   390→   390→        for step in 0..steps_per_walk {
   391→   391→   391→            // ── L7: Adaptive Walk Fusion ────────────────────
   392→   392→   392→            if !fusion_switched && dp_table.len() >= target_dp_count {
   393→   393→   393→                fusion_switched = true;
   394→   394→   394→                println!("  [FUSION] DP table filled ({} DPs) — shifting {} tame → wild",
   395→   395→   395→                         dp_table.len(), n_tame / 2);
   396→   396→   396→                // Convert half of tame walks to wild
   397→   397→   397→                let mut converted = 0;
   398→   398→   398→                for i in 0..all_walk_jacs.len() {
   399→   399→   399→                    if walk_is_tame[i] && converted < n_tame / 2 {
   400→   400→   400→                        // Restart this walk near target
   401→   401→   401→                        let offset = Fe::from_u64((converted + 100) as u64);
   402→   402→   402→                        let start_pt = self.target.add(&g.scalar_mul(&offset));
   403→   403→   403→                        all_walk_jacs[i] = start_pt.to_jacobian();
   404→   404→   404→                        all_walk_dists[i] = offset;
   405→   405→   405→                        walk_is_tame[i] = false;
   406→   406→   406→                        converted += 1;
   407→   407→   407→                    }
   408→   408→   408→                }
   409→   409→   409→                n_tame -= converted;
   410→   410→   410→                n_wild += converted;
   411→   411→   411→                println!("  [FUSION] Now: {} tame + {} wild", n_tame, n_wild);
   412→   412→   412→            }
   413→   413→   413→
   414→   414→   414→            // ── Step 1: Batch convert all walks (L2) ────────
   415→   415→   415→            let aff_points = batch_jac_to_affine(&all_walk_jacs);
   416→   416→   416→
   417→   417→   417→            // ── Step 2: DP check + GLV expansion (L1) ───────
   418→   418→   418→            for (i, aff) in aff_points.iter().enumerate() {
   419→   419→   419→                if aff.inf { continue; }
   420→   420→   420→                if aff.x.limbs[0] & dp_mask != 0 { continue; }
   421→   421→   421→
   422→   422→   422→                let x_bytes = aff.x.to_bytes();
   423→   423→   423→
   424→   424→   424→                // L4: Cubic Character check (skip for DP storage, use on collision)
   425→   425→   425→                let cubic_char = (aff.x.limbs[0] % 3) as u8;
   426→   426→   426→
   427→   427→   427→                if walk_is_tame[i] {
   428→   428→   428→                    // ══ TAME: Store GLV-expanded DPs ══
   429→   429→   429→                    let dist = all_walk_dists[i].clone();
   430→   430→   430→
   431→   431→   431→                    dp_table.entry(x_bytes).or_insert(DPEntry {
   432→   432→   432→                        distance: dist.clone(),
   433→   433→   433→                        glv_variant: 0,
   434→   434→   434→                        cubic_char,
   435→   435→   435→                    });
   436→   436→   436→
   437→   437→   437→                    let beta_x = BETA_FE.mul(&aff.x);
   438→   438→   438→                    dp_table.entry(beta_x.to_bytes()).or_insert(DPEntry {
   439→   439→   439→                        distance: dist.clone(),
   440→   440→   440→                        glv_variant: 1,
   441→   441→   441→                        cubic_char: (cubic_char + 1) % 3,
   442→   442→   442→                    });
   443→   443→   443→
   444→   444→   444→                    let beta2_x = BETA_FE.mul(&BETA_FE).mul(&aff.x);
   445→   445→   445→                    dp_table.entry(beta2_x.to_bytes()).or_insert(DPEntry {
   446→   446→   446→                        distance: dist,
   447→   447→   447→                        glv_variant: 2,
   448→   448→   448→                        cubic_char: (cubic_char + 2) % 3,
   449→   449→   449→                    });
   450→   450→   450→                } else {
   451→   451→   451→                    // ══ WILD: Check collision with 7-layer cascade ══
   452→   452→   452→                    if let Some(entry) = dp_table.get(&x_bytes) {
   453→   453→   453→                        collisions += 1;
   454→   454→   454→
   455→   455→   455→                        // L4: Cubic Character Oracle — reject 66% of false collisions
   456→   456→   456→                        if cubic_char != entry.cubic_char {
   457→   457→   457→                            cco_filtered += 1;
   458→   458→   458→                            continue;
   459→   459→   459→                        }
   460→   460→   460→
   461→   461→   461→                        // Try to recover key with remaining layers
   462→   462→   462→                        if let Some(k) = self.try_recover_7layer(
   463→   463→   463→                            &entry.distance,
   464→   464→   464→                            entry.glv_variant,
   465→   465→   465→                            &all_walk_dists[i],
   466→   466→   466→                            &rc_fe,
   467→   467→   467→                            &lambda_sq,
   468→   468→   468→                            &range_start,
   469→   469→   469→                            &range_end,
   470→   470→   470→                            &ens,
   471→   471→   471→                            &cco,
   472→   472→   472→                            &h160,
   473→   473→   473→                            &mut oracle_filtered,
   474→   474→   474→                            &mut ens_filtered,
   475→   475→   475→                            &mut cco_filtered,
   476→   476→   476→                            &mut h160_filtered,
   477→   477→   477→                        ) {
   478→   478→   478→                            found = true;
   479→   479→   479→                            found_k = Some(k);
   480→   480→   480→                            break;
   481→   481→   481→                        }
   482→   482→   482→                    }
   483→   483→   483→                }
   484→   484→   484→            }
   485→   485→   485→
   486→   486→   486→            if found { break; }
   487→   487→   487→
   488→   488→   488→            // ── Step 3: Advance all walks ───────────────────
   489→   489→   489→            for (i, aff) in aff_points.iter().enumerate() {
   490→   490→   490→                let si = hash_step(aff, n_steps);
   491→   491→   491→                all_walk_jacs[i] = all_walk_jacs[i].add_affine(&step_points[si]);
   492→   492→   492→                all_walk_dists[i] = all_walk_dists[i].add_mod_n(&step_scalars[si]);
   493→   493→   493→            }
   494→   494→   494→
   495→   495→   495→            total_steps += N_WALKS as u64;
   496→   496→   496→
   497→   497→   497→            if step > 0 && step % 500_000 == 0 {
   498→   498→   498→                let elapsed = start.elapsed().as_secs_f64();
   499→   499→   499→                let rate = total_steps as f64 / elapsed;
   500→   500→   500→                println!("    Step {}: {} total | {} DPs | {} coll | ENS:{} CCO:{} H160:{} | {:.0}/s",
   501→   501→   501→                         step, total_steps, dp_table.len(), collisions,
   502→   502→   502→                         ens_filtered, cco_filtered, h160_filtered, rate);
   503→   503→   503→            }
   504→   504→   504→        }
   505→   505→   505→
   506→   506→   506→        let elapsed_ms = start.elapsed().as_millis() as u64;
   507→   507→   507→        let dp_count = dp_table.len() as u64;
   508→   508→   508→
   509→   509→   509→        if found {
   510→   510→   510→            PrismResult {
   511→   511→   511→                found: true, k: found_k, steps: total_steps,
   512→   512→   512→                collisions, dp_count, oracle_filtered,
   513→   513→   513→                ens_filtered, cco_filtered, h160_filtered, elapsed_ms,
   514→   514→   514→            }
   515→   515→   515→        } else {
   516→   516→   516→            println!("\n  [PRISM V2] Search complete: {} steps, {} DPs, {} collisions",
   517→   517→   517→                     total_steps, dp_count, collisions);
   518→   518→   518→            println!("  [PRISM V2] Layer rejections: ENS={}, CCO={}, H160={}, SHA={}",
   519→   519→   519→                     ens_filtered, cco_filtered, h160_filtered, oracle_filtered);
   520→   520→   520→            PrismResult {
   521→   521→   521→                found: false, k: None, steps: total_steps,
   522→   522→   522→                collisions, dp_count, oracle_filtered,
   523→   523→   523→                ens_filtered, cco_filtered, h160_filtered, elapsed_ms,
   524→   524→   524→            }
   525→   525→   525→        }
   526→   526→   526→    }

   528→   528→   528→    // ════════════════════════════════════════════════════════
   529→   529→   529→    //  7-LAYER CASCADE KEY RECOVERY
   530→   530→   530→    // ════════════════════════════════════════════════════════
   531→   531→   531→    //
   532→   532→   532→    // Pipeline: GLV recovery → L3(ENS) → L5(H160) → L6(SHA) → verify
   533→   533→   533→
   534→   534→   534→    fn try_recover_7layer(
   535→   535→   535→        &self,
   536→   536→   536→        tame_dist: &Fe,
   537→   537→   537→        glv_variant: u8,
   538→   538→   538→        wild_dist: &Fe,
   539→   539→   539→        rc_fe: &Fe,
   540→   540→   540→        lambda_sq: &Fe,
   541→   541→   541→        range_start: &BigUint,
   542→   542→   542→        range_end: &BigUint,
   543→   543→   543→        ens: &EisensteinNormSieve,
   544→   544→   544→        _cco: &CubicCharacterOracle,
   545→   545→   545→        h160: &Hash160Oracle,
   546→   546→   546→        oracle_filtered: &mut u64,
   547→   547→   547→        ens_filtered: &mut u64,
   548→   548→   548→        cco_filtered: &mut u64,
   549→   549→   549→        h160_filtered: &mut u64,
   550→   550→   550→    ) -> Option<BigUint> {
   551→   551→   551→        let g = Point::generator();
   552→   552→   552→
   553→   553→   553→        let base_tame = rc_fe.add_mod_n(tame_dist);
   554→   554→   554→
   555→   555→   555→        let tame_scalar = match glv_variant {
   556→   556→   556→            0 => base_tame,
   557→   557→   557→            1 => base_tame.mul_mod_n(&LAMBDA_FE),
   558→   558→   558→            2 => base_tame.mul_mod_n(lambda_sq),
   559→   559→   559→            _ => base_tame,
   560→   560→   560→        };
   561→   561→   561→
   562→   562→   562→        for &sign_scalar in &[tame_scalar, tame_scalar.neg_mod_n()] {
   563→   563→   563→            let k_fe = sign_scalar.sub_mod_n(wild_dist);
   564→   564→   564→            let k_big = k_fe.to_biguint();
   565→   565→   565→
   566→   566→   566→            // ── Range check ──
   567→   567→   567→            if k_big < *range_start || k_big >= *range_end { continue; }
   568→   568→   568→
   569→   569→   569→            // ── L3: Eisenstein Norm Sieve ──
   570→   570→   570→            // O(1) BigUint arithmetic — rejects ~99.8% of false collisions
   571→   571→   571→            if !ens.check(&k_big) {
   572→   572→   572→                *ens_filtered += 1;
   573→   573→   573→                continue;
   574→   574→   574→            }
   575→   575→   575→
   576→   576→   576→            // ── L5: Hash160 Oracle ──
   577→   577→   577→            // Compute k*G, get x, check Hash160 matches target
   578→   578→   578→            // This is expensive but still cheaper than full verification
   579→   579→   579→            // We do it BEFORE the SHA oracle because SHA is cheaper
   580→   580→   580→            let q = g.scalar_mul(&k_fe);
   581→   581→   581→            if q.inf { continue; }
   582→   582→   582→
   583→   583→   583→            // Quick x comparison first (free)
   584→   584→   584→            if q.x == self.target.x {
   585→   585→   585→                if q.y == self.target.y || q.y == self.target.y.neg_mod_p() {
   586→   586→   586→                    println!("  *** KEY FOUND: 0x{:x} ***", k_big);
   587→   587→   587→                    return Some(k_big);
   588→   588→   588→                }
   589→   589→   589→            }
   590→   590→   590→
   591→   591→   591→            // L5: Hash160 check (16 bits of filter)
   592→   592→   592→            let y_par = (q.y.limbs[0] & 1) as u8;
   593→   593→   593→            if !h160.check(&q.x.to_bytes(), y_par) {
   594→   594→   594→                *h160_filtered += 1;
   595→   595→   595→                continue;
   596→   596→   596→            }
   597→   597→   597→
   598→   598→   598→            // ── L6: SHA-256 Oracle ──
   599→   599→   599→            if let Some(ref oracle) = self.oracle {
   600→   600→   600→                if !oracle.check_x(&q.x.to_bytes()) {
   601→   601→   601→                    *oracle_filtered += 1;
   602→   602→   602→                    continue;
   603→   603→   603→                }
   604→   604→   604→            }
   605→   605→   605→
   606→   606→   606→            // ── L4: CCO double-check ──
   607→   607→   607→            if (q.x.limbs[0] % 3) as u8 != _cco.target_char {
   608→   608→   608→                *cco_filtered += 1;
   609→   609→   609→                continue;
   610→   610→   610→            }
   611→   611→   611→
   612→   612→   612→            // Full verification (redundant but safe)
   613→   613→   613→            if q.x == self.target.x &&
   614→   614→   614→               (q.y == self.target.y || q.y == self.target.y.neg_mod_p()) {
   615→   615→   615→                println!("  *** KEY FOUND: 0x{:x} ***", k_big);
   616→   616→   616→                return Some(k_big);
   617→   617→   617→            }
   618→   618→   618→        }
   619→   619→   619→
   620→   620→   620→        None
   621→   621→   621→    }

   623→   623→   623→    /// Self-test: generate random key in range, find it with PRISM VORTEX V2
   624→   624→   624→    pub fn selftest(range_bits: u32) -> PrismResult {
   625→   625→   625→        let range_bits = std::cmp::min(range_bits, 40);
   626→   626→   626→        let g = Point::generator();
   627→   627→   627→
   628→   628→   628→        let mut seed = range_bits as u64 * 0x5851F42D4C957F2D;
   629→   629→   629→        let mut next_rand = || -> u64 {
   630→   630→   630→            seed = seed.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
   631→   631→   631→            seed
   632→   632→   632→        };
   633→   633→   633→
   634→   634→   634→        let range_start = BigUint::from(1u64) << (range_bits - 1);
   635→   635→   635→        let offset = next_rand() % 1000;
   636→   636→   636→        let k_big = range_start.clone() + offset;
   637→   637→   637→        let k_fe = Fe::from_biguint_mod_n(&k_big);
   638→   638→   638→

   640→   640→   640→        println!("  [SELFTEST] Range: [2^{}, 2^{})", range_bits - 1, range_bits);
   641→   641→   641→

   643→   643→   643→        if !target.is_on_curve() {
   644→   644→   644→            println!("  [SELFTEST] ERROR: target not on curve!");

   651→   651→   651→
   652→   652→   652→        let x_bytes = target.x.to_bytes();
   653→   653→   653→        let y_is_odd = target.y.limbs[0] & 1 == 1;
   654→   654→   654→        let mut pubkey_bytes = [0u8; 33];

   660→   660→   660→            0..=25 => 10_000_000,
   661→   661→   661→            26..=30 => 50_000_000,
   662→   662→   662→            31..=35 => 100_000_000,
   663→   663→   663→            36..=40 => 500_000_000,
   664→   664→   664→            _ => 1_000_000_000,
   665→   665→   665→        };
   666→   666→   666→
   667→   667→   667→        let solver = PrismVortex::new(range_bits, target, Some(oracle));
   668→   668→   668→        let result = solver.solve(max_steps);
   669→   669→   669→
   670→   670→   670→        if result.found {
   671→   671→   671→            if let Some(ref k_found) = result.k {
   672→   672→   672→                let match_ok = k_found == &k_big;
   673→   673→   673→                println!("\n  ╔══════════════════════════════════════╗");
   674→   674→   674→                println!("  ║  PRISM VORTEX V2: KEY FOUND!          ║");
   675→   675→   675→                println!("  ║  k_found = 0x{:x}", k_found);
   676→   676→   676→                println!("  ║  k_real  = 0x{:x}", k_big);
   677→   677→   677→                println!("  ║  MATCH: {}                    ║", match_ok);
   678→   678→   678→                println!("  ║  Layers: ENS={} CCO={} H160={} SHA={}",
   679→   679→   679→                         result.ens_filtered, result.cco_filtered,
   680→   680→   680→                         result.h160_filtered, result.oracle_filtered);
   681→   681→   681→                println!("  ╚══════════════════════════════════════╝");
   682→   682→   682→            }
   683→   683→   683→        } else {
   684→   684→   684→            println!("\n  [SELFTEST] Not found in {} steps", result.steps);
   685→   685→   685→            println!("  [SELFTEST] Need ~2^{} steps for {}-bit range",
   686→   686→   686→                     range_bits / 2, range_bits);
   687→   687→   687→        }
   688→   688→   688→
   689→   689→   689→        result
   690→   690→   690→    }
   691→   691→   691→}

   694→   694→   694→// BATCH JACOBIAN → AFFINE (Montgomery's trick)
   695→   695→   695→// ============================================================
   696→   696→   696→
   697→   697→   697→fn batch_jac_to_affine(points: &[JacobianPoint]) -> Vec<Point> {
   698→   698→   698→    let n = points.len();
   699→   699→   699→    if n == 0 { return Vec::new(); }
   700→   700→   700→    if n == 1 { return vec![points[0].to_affine()]; }
   701→   701→   701→
   702→   702→   702→    let mut prefix = Vec::with_capacity(n);
   703→   703→   703→    prefix.push(points[0].z);
   704→   704→   704→    for i in 1..n {
   705→   705→   705→        prefix.push(prefix[i - 1].mul(&points[i].z));
   706→   706→   706→    }
   707→   707→   707→
   708→   708→   708→    let inv_all = prefix[n - 1].modinv();
   709→   709→   709→
   710→   710→   710→    let mut z_inv = vec![Fe::ZERO; n];
   711→   711→   711→    let mut acc = inv_all;
   712→   712→   712→    for i in (1..n).rev() {
   713→   713→   713→        z_inv[i] = acc.mul(&prefix[i - 1]);
   714→   714→   714→        acc = acc.mul(&points[i].z);
   715→   715→   715→    }
   716→   716→   716→    z_inv[0] = acc;
   717→   717→   717→
   718→   718→   718→    points.iter().enumerate().map(|(i, pt)| {
   719→   719→   719→        if pt.z.is_zero() {
   720→   720→   720→            Point::infinity()
   721→   721→   721→        } else {
   722→   722→   722→            let zi = z_inv[i];
   723→   723→   723→            let zi2 = zi.mul(&zi);
   724→   724→   724→            let zi3 = zi2.mul(&zi);
   725→   725→   725→            Point {
   726→   726→   726→                x: pt.x.mul(&zi2),
   727→   727→   727→                y: pt.y.mul(&zi3),
   728→   728→   728→                inf: false,
   729→   729→   729→            }
   730→   730→   730→        }
   731→   731→   731→    }).collect()
   732→   732→   732→}
