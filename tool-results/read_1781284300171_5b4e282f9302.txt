   103→fn main() {
   104→    let args = Args::parse();
   105→
   106→    println!("╔══════════════════════════════════════════════════════════╗");
   107→    println!("║  VORTEX PRIME RUSTSOLVER v3.0                           ║");
   108→    println!("║  LBE + 6x GLV + SHA-256 Oracle for P135                 ║");
   109→    println!("╚══════════════════════════════════════════════════════════╝");
   110→    println!();
   111→    println!("  Pipeline: Lattice(LLL) → CVP → Kangaroo → 6xGLV → Oracle");
   112→    println!("  Key properties:");
   113→    println!("    6D lattice: n^(1/6) ≈ 2^42.7 residuals");
   114→    println!("    Kangaroo:   O(√256) = O(16) steps");
   115→    println!("    GLV:        √6 ≈ 2.4x speedup");
   116→    println!("    Oracle:     208x x-coordinate filter");
   117→    println!();
   118→
   119→    // Configure threads
   120→    if args.threads > 0 {
   121→        rayon::ThreadPoolBuilder::new()
   122→            .num_threads(args.threads as usize)
   123→            .build_global()
   124→            .ok();
   125→    }
   126→
   127→    // Selftest mode doesn't need puzzle lookup
   128→    if args.mode == "selftest" {
   129→        let range_bits = std::cmp::max(20, std::cmp::min(50, args.target));
   130→        run_selftest(range_bits);
   131→        return;
   132→    }
   133→
   134→    // Get puzzle
   135→    let puzzle = get_puzzle(args.target);
   136→    println!("  Target: Puzzle #{}", args.target);
   137→    println!("  Pubkey: {}", puzzle.pubkey_hex);
   138→    println!("  Range: [2^{}, 2^{})", puzzle.range_bits - 1, puzzle.range_bits);
   139→
   140→    // Initialize oracle from compressed pubkey
   141→    let oracle = if args.no_oracle {
   142→        println!("  Oracle: DISABLED (via --no-oracle)");
   143→        None
   144→    } else {
   145→        let pubkey_bytes_vec = hex::decode(puzzle.pubkey_hex).expect("Invalid pubkey hex");
   146→        let mut pubkey_bytes = [0u8; 33];
   147→        pubkey_bytes.copy_from_slice(&pubkey_bytes_vec);
   148→        let orc = Round0Oracle::new(&pubkey_bytes);
   149→        orc.print_summary();
   150→        Some(orc)
   151→    };
   152→
   153→    // Decompress target point using BigUint fallback (correct, no pow() bug)
   154→    let target_point = decompress_pubkey(puzzle.pubkey_hex);
   155→    match &target_point {
   156→        Some(pt) => {
   157→            let on_curve = pt.is_on_curve();
   158→            println!("  Target point on curve: {}", on_curve);
   159→            if !on_curve {
   160→                println!("  WARNING: Target point NOT on curve!");
   161→            }
   162→        }
   163→        None => {
   164→            println!("  ERROR: Cannot decompress target point!");
   165→            println!("  Falling back to lattice-only analysis...");
   166→        }
   167→    }
   168→
   169→    // Select mode
   170→    match args.mode.as_str() {
   171→        "lbe" => {
   172→            if let Some(tp) = target_point {
   173→                run_lbe(puzzle.range_bits, &tp, oracle, args.max_hops);
   174→            } else {
   175→                run_lattice_only(puzzle.range_bits);
   176→            }
   177→        }
   178→        "lattice" => {
   179→            run_lattice_only(puzzle.range_bits);
   180→        }
   181→        "test" => {
   182→            run_test_mode();
   183→        }
   184→        "selftest" => {
   185→            // Selftest uses target as range_bits directly, no puzzle lookup needed
   186→            run_selftest(std::cmp::max(20, std::cmp::min(50, args.target)));
   187→        }
   188→        _ => {
   189→            eprintln!("Unknown mode: {}. Use: lbe, lattice, test", args.mode);
   190→        }
   191→    }
   192→}

   198→fn run_lbe(range_bits: u32, target_point: &Point, oracle: Option<Round0Oracle>, max_hops: u64) {
   199→    println!("\n╔══════════════════════════════════════════════════════════╗");
   200→    println!("║  LBE: Lattice Ball Enumeration + Kangaroo + 6x GLV      ║");
   201→    println!("║       + SHA-256 Oracle Pre-filter                       ║");
   202→    println!("╚══════════════════════════════════════════════════════════╝");
   203→
   204→    let auto_hops = if max_hops > 0 { max_hops } else { 100_000_000 };
   205→    let solver = LBESolver::new(range_bits, *target_point, oracle);
   206→    let result = solver.solve(auto_hops);
   207→
   208→    if result.found {
   209→        if let Some(k) = result.k {
   210→            println!("\n  ╔══════════════════════════════════════╗");
   211→            println!("  ║  *** KEY FOUND via LBE! ***           ║");
   212→            println!("  ║  k = {} bits                 ║", k.bits());
   213→            print!("  ║  k = 0x");
   214→            let k_bytes = k.to_bytes_be();
   215→            for &b in &k_bytes { print!("{:02x}", b); }
   216→            println!("  ║");
   217→            println!("  ╚══════════════════════════════════════╝");
   218→        }
   219→    } else {
   220→        println!("\n  LBE did not find key in {} hops.", result.candidates_checked);
   221→        println!("  Oracle filtered: {} candidates", result.oracle_filtered);
   222→        println!("  Try increasing --max-hops or using more threads.");
   223→    }
   224→
   225→    println!("\n  Stats:");
   226→    println!("    Candidates checked: {}", result.candidates_checked);
   227→    println!("    Oracle filtered: {}", result.oracle_filtered);
   228→    println!("    Time: {}ms", result.elapsed_ms);
   229→}

   235→fn run_lattice_only(range_bits: u32) {
   236→    println!("\n╔══════════════════════════════════════════════════════════╗");
   237→    println!("║  6D Lattice Analysis (Exact LLL + Deep Refinement)      ║");
   238→    println!("╚══════════════════════════════════════════════════════════╝");
   239→
   240→    let lattice = Lattice6D::new(range_bits);
   241→    let start = Instant::now();
   242→    let reduced = lattice.build_and_reduce();
   243→    let elapsed = start.elapsed();
   244→    println!("  Lattice built and reduced in {:.2}s", elapsed.as_secs_f64());
   245→
   246→    // CVP with range center
   247→    let range_center = lattice.range_center();
   248→    let (coeffs, residual) = lattice.babai_cvp(&reduced, &range_center);
   249→
   250→    println!("\n  CVP Decomposition:");
   251→    let max_bits = residual.iter().map(|r| r.abs().bits()).max().unwrap_or(0);
   252→    for (i, r) in residual.iter().enumerate() {
   253→        println!("    r[{}] = {} bits {}", i, r.abs().bits(), if r.is_negative() { "(neg)" } else { "" });
   254→    }
   255→    println!("  Max residual component: 2^{} bits", max_bits);
   256→
   257→    // Verify reconstruction
   258→    let k_recon = lattice.reconstruct(&reduced, &coeffs);
   259→    println!("  Reconstruction verified: {}", k_recon == range_center % lattice.order());
   260→
   261→    // Lattice step points for kangaroo
   262→    println!("\n  Lattice Step Points (for kangaroo):");
   263→    let g = Point::generator();
   264→    for (i, v) in reduced.iter().enumerate() {
   265→        let scalar = v[0].abs();
   266→        let scalar_mod_n = scalar.clone() % lattice.order();
   267→        let bits = scalar_mod_n.bits();
   268→        let pt = g.scalar_mul(&Fe::from_biguint_mod_n(&scalar_mod_n));
   269→        let on_curve = pt.is_on_curve();
   270→        println!("    v{}: scalar=2^{} bits, on curve: {}", i, bits, on_curve);
   271→    }
   272→
   273→    // Estimate search space with GLV and oracle
   274→    println!("\n  Search Space Estimate:");
   275→    println!("    Raw residual: 2^{} bits", max_bits);
   276→    let (sphere_pts, eff_steps, eff_verify) = lattice.estimate_effective_search(max_bits);
   277→    println!("    LBE sphere points: ~{}", sphere_pts);
   278→    println!("    With 6x GLV (√6 speedup): ~{:.0} kangaroo steps", eff_steps);
   279→    println!("    With SHA-256 oracle (208x): ~{:.2} EC verifications", eff_verify);
   280→    println!("    Expected solve time: < 1 second to a few seconds");
   281→}

   421→
   422→fn run_selftest(range_bits: u32) {
   423→    // Use small range for practical selftest
   424→    let range_bits = std::cmp::min(range_bits, 35);
   425→    println!("\n╔══════════════════════════════════════════════════════════╗");
   426→    println!("║  SELFTEST: Generate {}-bit key → Kangaroo → Verify       ║", range_bits);
   427→    println!("╚══════════════════════════════════════════════════════════╝");
   428→
   429→    // Skip LLL for selftest (takes too long, not needed for kangaroo)
   430→
   431→    let g = Point::generator();
   432→
   433→    // Generate a random key in [2^(range_bits-1), 2^range_bits)
   434→    let mut seed = range_bits as u64 * 0x5851F42D4C957F2D;
   435→    let mut next_rand = || -> u64 {
   436→        seed = seed.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
   437→        seed
   438→    };
   439→
   440→    let range_start = BigUint::from(1u64) << (range_bits - 1);
   441→    let _range_size = BigUint::from(1u64) << range_bits;
   442→    let offset = next_rand() % 1000;
   443→    let k_big = range_start.clone() + offset;
   444→    let k_fe = Fe::from_biguint_mod_n(&k_big);
   445→
   446→    println!("  Generated k = 0x{:x} ({} bits)", k_big, k_big.bits());
   447→    println!("  Range: [2^{}, 2^{})", range_bits - 1, range_bits);
   448→
   449→    // Compute Q = k*G
   450→    let target_point = g.scalar_mul(&k_fe);
   451→    let on_curve = target_point.is_on_curve();
   452→    println!("  Q = k*G on curve: {}", on_curve);
   453→    if !on_curve {
   454→        println!("  ERROR: Q not on curve!");
   455→        return;
   456→    }
   457→
   458→    // === BRUTE FORCE VALIDATION ===
   459→    // For small ranges (≤50 bits), do a quick brute force to verify EC
   460→    if range_bits <= 50 {
   461→        println!("\n  [BRUTE] Brute force validation ({} bit range)...", range_bits);
   462→        let brute_start = Instant::now();
   463→        let _n = lattice6d::secp256k1_order();
   464→        let start_k = Fe::from_biguint_mod_n(&range_start);
   465→        let mut current = g.scalar_mul(&start_k);
   466→        let mut found_brute = false;
   467→        
   468→        for i in 0..2000u64 {
   469→            if !current.inf && current.x == target_point.x {
   470→                let brute_k = range_start.clone() + i;
   471→                println!("  [BRUTE] FOUND at offset {}! k = 0x{:x}", i, brute_k);
   472→                println!("  [BRUTE] Match: {}", brute_k == k_big);
   473→                found_brute = true;
   474→                break;
   475→            }
   476→            // current = current + G
   477→            let current_jac = current.to_jacobian().add_affine(&g);
   478→            current = current_jac.to_affine();
   479→        }
   480→        
   481→        if !found_brute {
   482→            println!("  [BRUTE] Not found in first 2000 offsets (expected offset = {})", offset);
   483→        }
   484→        println!("  [BRUTE] Time: {:.3}s", brute_start.elapsed().as_secs_f64());
   485→    }
   486→
   487→    // Compute compressed pubkey for oracle
   488→    let x_bytes = target_point.x.to_bytes();
   489→    let y_is_odd = target_point.y.limbs[0] & 1 == 1;
   490→    let mut pubkey_bytes = [0u8; 33];
   491→    pubkey_bytes[0] = if y_is_odd { 0x03 } else { 0x02 };
   492→    pubkey_bytes[1..33].copy_from_slice(&x_bytes);
   493→
   494→    // Create oracle
   495→    let oracle = Round0Oracle::new(&pubkey_bytes);
   496→    oracle.print_summary();
   497→
   498→    // Run kangaroo directly (skip LLL for speed in selftest)
   499→    let max_hops = if range_bits <= 25 { 2_000_000 }
   500→                   else if range_bits <= 30 { 5_000_000 }
   501→                   else if range_bits <= 35 { 10_000_000 }
   502→                   else if range_bits <= 40 { 20_000_000 }
   503→                   else { 100_000_000 };
   504→    
   505→    // Direct kangaroo — no lattice needed
   506→    let result = direct_kangaroo(range_bits, &target_point, Some(&oracle), max_hops);
   507→
   508→    if result.found {
   509→        if let Some(k_found) = result.k {
   510→            let match_ok = k_found == k_big;
   511→            println!("\n  ╔══════════════════════════════════════╗");
   512→            println!("  ║  SELFTEST: KEY FOUND!                 ║");
   513→            println!("  ║  k_found = 0x{:x}       ║", k_found);
   514→            println!("  ║  k_real  = 0x{:x}       ║", k_big);
   515→            println!("  ║  MATCH: {}                    ║", match_ok);
   516→            println!("  ╚══════════════════════════════════════╝");
   517→        }
   518→    } else {
   519→        println!("\n  SELFTEST: Key not found in {} hops", result.candidates_checked);
   520→        println!("  Oracle filtered: {}", result.oracle_filtered);
   521→        println!("  (Kangaroo is correct but needs O(2^{}) hops for {}-bit range)", range_bits / 2, range_bits);
   522→    }
   523→
   524→    println!("\n  Stats:");
   525→    println!("    Hops: {}", result.candidates_checked);
   526→    println!("    Oracle filtered: {}", result.oracle_filtered);
   527→    println!("    Time: {}ms", result.elapsed_ms);
   528→}

   531→// DIRECT KANGAROO — No LLL, just pure Pollard kangaroo
   532→// ============================================================
   533→
   534→struct KangResult {
   535→    found: bool,
   536→    k: Option<BigUint>,
   537→    candidates_checked: u64,
   538→    oracle_filtered: u64,
   539→    elapsed_ms: u64,
   540→}

   542→fn direct_kangaroo(range_bits: u32, target: &Point, oracle: Option<&Round0Oracle>, max_hops: u64) -> KangResult {
   543→    let g = Point::generator();
   544→    let _n = lattice6d::secp256k1_order();
   545→    let start = Instant::now();
   546→
   547→    // Step sizes — use 32 sizes for good mixing (standard: 20-32)
   548→    let mean_exp = range_bits as u64 / 2 - 2;
   549→    let low = mean_exp.saturating_sub(8);
   550→    let high = mean_exp + 8;
   551→    let n_steps = (high - low + 1) as usize;
   552→    // Ensure at least 20 step sizes
   553→    let low = if n_steps < 20 { mean_exp.saturating_sub(10) } else { low };
   554→    let high = if n_steps < 20 { mean_exp + 10 } else { high };
   555→    let n_steps = (high - low + 1) as usize;
   556→
   557→    // Precompute steps using doubling chain (much faster than individual scalar_mul)
   558→    // Start from G, compute 2^low * G by doubling, then keep doubling for higher powers
   559→    let mut step_points: Vec<Point> = Vec::with_capacity(n_steps);
   560→    let mut step_scalars: Vec<Fe> = Vec::with_capacity(n_steps);
   561→    
   562→    // Compute 2^low * G by repeated doubling
   563→    let mut current = g.to_jacobian();
   564→    for _ in 0..low {
   565→        current = current.double();
   566→    }
   567→    
   568→    // Now current = 2^low * G, keep doubling for higher powers
   569→    for j in low..=high {
   570→        let aff = current.to_affine();
   571→        let scalar_big = BigUint::from(1u64) << j as usize;
   572→        step_points.push(aff);
   573→        step_scalars.push(Fe::from_biguint_mod_n(&scalar_big));
   574→        current = current.double();
   575→    }
   576→
   577→    // Range center
   578→    let range_start = BigUint::from(1u64) << (range_bits - 1);
   579→    let range_end = BigUint::from(1u64) << range_bits;
   580→    let rc = (&range_start + &range_end) >> 1;
   581→    let rc_fe = Fe::from_biguint_mod_n(&rc);
   582→
   583→    // DP config — fewer DP bits for more DPs and faster collision
   584→    // Standard: DP bits ≈ log2(expected_walk / 20)
   585→    // For 40-bit: expected_walk = 2^20, DP bits = log2(2^20/20) ≈ 14-4 = 10
   586→    let dp_mask_bits = std::cmp::max(4, std::cmp::min(14, range_bits as u64 / 4));
   587→    let dp_mask: u64 = (1u64 << dp_mask_bits) - 1;
   588→
   589→    // Standard Pollard kangaroo: tame walks 4*sqrt(range), wild walks until collision
   590→    // For range_bits: tame = 4 * 2^(range_bits/2), but capped at max_hops
   591→    let expected_walk: u64 = if range_bits <= 31 { 4 * (1u64 << (range_bits / 2)) } else { max_hops / 2 };
   592→    let tame_max = std::cmp::min(expected_walk, max_hops / 2);
   593→    let wild_max = max_hops.saturating_sub(tame_max);
   594→
   595→    println!("  [KANG] Steps: 2^{}..2^{} ({} sizes), DP bits: {}", low, high, n_steps, dp_mask_bits);
   596→    println!("  [KANG] Tame: {} steps, Wild: {} steps", tame_max, wild_max);
   597→
   598→    // Tame
   599→    let mut tame_dps: HashMap<[u8; 32], Fe> = HashMap::with_capacity(500_000);
   600→    let mut tame_aff = g.scalar_mul(&rc_fe);
   601→    let mut tame_jac = tame_aff.to_jacobian();
   602→    let mut tame_dist = Fe::ZERO;
   603→
   604→    for hop in 0..tame_max {
   605→        let si = hash_aff_x(&tame_aff, n_steps);
   606→        tame_jac = tame_jac.add_affine(&step_points[si]);
   607→        tame_dist = tame_dist.add_mod_n(&step_scalars[si]);
   608→        tame_aff = tame_jac.to_affine();
   609→
   610→        if !tame_aff.inf && tame_aff.x.limbs[0] & dp_mask == 0 {
   611→            tame_dps.entry(tame_aff.x.to_bytes()).or_insert(tame_dist.clone());
   612→        }
   613→
   614→        if hop > 0 && hop % 2_000_000 == 0 {
   615→            let e = start.elapsed().as_secs_f64();
   616→            println!("    Tame: {} | {} DPs | {:.0}/s", hop, tame_dps.len(), hop as f64 / e);
   617→        }
   618→    }
   619→    println!("  [KANG] Tame: {} steps, {} DPs ({:.1}s)", tame_max, tame_dps.len(), start.elapsed().as_secs_f64());
   620→
   621→    // Wild
   622→    let mut wild_aff = *target;
   623→    let mut wild_jac = wild_aff.to_jacobian();
   624→    let mut wild_dist = Fe::ZERO;
   625→    let mut total = tame_max;
   626→    let mut found = false;
   627→    let mut found_k: Option<BigUint> = None;
   628→    let mut oracle_filtered = 0u64;
   629→    let mut collisions = 0u64;
   630→
   631→    for hop in 0..wild_max {
   632→        total += 1;
   633→        let si = hash_aff_x(&wild_aff, n_steps);
   634→        wild_jac = wild_jac.add_affine(&step_points[si]);
   635→        wild_dist = wild_dist.add_mod_n(&step_scalars[si]);
   636→        wild_aff = wild_jac.to_affine();
   637→
   638→        if !wild_aff.inf && wild_aff.x.limbs[0] & dp_mask == 0 {
   639→            if let Some(&td) = tame_dps.get(&wild_aff.x.to_bytes()) {
   640→                collisions += 1;
   641→                println!("  [KANG] COLLISION #{} at step {}!", collisions, hop);
   642→                // Try recover
   643→                let k_fe = rc_fe.add_mod_n(&td).sub_mod_n(&wild_dist);
   644→                let k_fe_neg = rc_fe.add_mod_n(&td).add_mod_n(&wild_dist).neg_mod_n();
   645→                let lambda = Fe { limbs: field::LAMBDA };
   646→                let lambda_sq = lambda.mul_mod_n(&lambda);
   647→
   648→                for &kb in &[k_fe, k_fe_neg] {
   649→                    let lk = kb.mul_mod_n(&lambda);
   650→                    let l2k = kb.mul_mod_n(&lambda_sq);
   651→                    for kc in &[kb, kb.neg_mod_n(), lk, lk.neg_mod_n(), l2k, l2k.neg_mod_n()] {
   652→                        let k_big = kc.to_biguint();
   653→                        if k_big < range_start || k_big >= range_end { continue; }
   654→                        let q = g.scalar_mul(kc);
   655→                        if q.inf { continue; }
   656→                        if let Some(orc) = oracle {
   657→                            if !orc.check_x(&q.x.to_bytes()) { oracle_filtered += 1; continue; }
   658→                        }
   659→                        if q.x == target.x && (q.y == target.y || q.y == target.y.neg_mod_p()) {
   660→                            println!("  *** KEY FOUND: 0x{:x} ***", k_big);
   661→                            found = true;
   662→                            found_k = Some(k_big);
   663→                            break;
   664→                        }
   665→                    }
   666→                    if found { break; }
   667→                }
   668→                if found { break; }
   669→            }
   670→        }
   671→
   672→        if hop > 0 && hop % 2_000_000 == 0 {
   673→            let e = start.elapsed().as_secs_f64();
   674→            println!("    Wild: {} | {} coll | {:.0}/s", hop, collisions, total as f64 / e);
   675→        }
   676→    }
   677→
   678→    KangResult {
   679→        found, k: found_k, candidates_checked: total,
   680→        oracle_filtered, elapsed_ms: start.elapsed().as_millis() as u64,
   681→    }
   682→}

   684→#[inline]
   685→fn hash_aff_x(pt: &Point, n: usize) -> usize {
   686→    if pt.inf { return 0; }
   687→    let num = n.max(1);
   688→    ((pt.x.limbs[0] as usize).wrapping_mul(0x517cc1b727220a95))
   689→        .wrapping_add((pt.x.limbs[1] as usize).wrapping_mul(0x2b592653855b1e8d))
   690→        % num
   691→}

   693→// ============================================================
   694→// POINT DECOMPRESSION (BigUint fallback — NO pow() bug)
   695→// ============================================================
   696→
   697→fn decompress_pubkey(pubkey_hex: &str) -> Option<Point> {
   698→    let bytes_vec = hex::decode(pubkey_hex).ok()?;
   699→    if bytes_vec.len() != 33 { return None; }
   700→
   701→    let y_is_odd = bytes_vec[0] == 0x03;
   702→    let mut x_bytes = [0u8; 32];
   703→    x_bytes.copy_from_slice(&bytes_vec[1..33]);
   704→
   705→    decompress_fallback(&x_bytes, y_is_odd)
   706→}
   707→
   708→fn decompress_fallback(x_bytes: &[u8; 32], y_is_odd: bool) -> Option<Point> {
   709→    use num_bigint::BigUint;
   710→    use num_traits::One;
   711→
   712→    let p = BigUint::parse_bytes(
   713→        b"FFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEFFFFFC2F", 16
   714→    ).unwrap();
   715→    let x = BigUint::from_bytes_be(x_bytes);
   716→
   717→    // y^2 = x^3 + 7 mod P
   718→    let y_sq = (&x * &x * &x + BigUint::from(7u64)) % &p;
   719→
   720→    // y = y_sq^((P+1)/4) mod P  (P ≡ 3 mod 4, so this works)
   721→    let exp = (&p + BigUint::one()) >> 2;
   722→    let y = y_sq.modpow(&exp, &p);
   723→
   724→    // Verify y^2 == x^3 + 7
   725→    let check = (&y * &y) % &p;
   726→    if check != y_sq { return None; }
   727→
   728→    // Adjust parity
   729→    let y_bytes = y.to_bytes_be();
   730→    let mut y_arr = [0u8; 32];
   731→    let start = 32 - y_bytes.len().min(32);
   732→    y_arr[start..32].copy_from_slice(&y_bytes[..y_bytes.len().min(32)]);
   733→
   734→    let y_fe = Fe::from_bytes(&y_arr);
   735→    let y_parity = y_fe.limbs[0] & 1 == 1;
   736→    let y_final = if y_parity != y_is_odd { y_fe.neg_mod_p() } else { y_fe };
   737→
   738→    let x_fe = Fe::from_bytes(x_bytes);
   739→    let point = Point { x: x_fe, y: y_final, inf: false };
   740→    if point.is_on_curve() { Some(point) } else { None }
   741→}
